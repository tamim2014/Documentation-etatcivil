function captureSousMenu(prfctr) {
    if (prfctr === "") {
        document.getElementById("yivawo").innerHTML = "";
        return;
    }

    const xmlhttp = new XMLHttpRequest();
    xmlhttp.open("GET", "SERVEUR/lectureBD_afficherNaissance.php?pr=" + prfctr, true);
    xmlhttp.send();

    xmlhttp.onreadystatechange = function () {
        if (xmlhttp.readyState === 4 && xmlhttp.status === 200) {
            document.getElementById("yivawo").innerHTML = xmlhttp.responseText;
        }
    };
}

function captureSousMenu(prfctr) {
    const cible = document.getElementById("yivawo");

    if (prfctr === "") {
        cible.innerHTML = "";
        return;
    }

    fetch(`SERVEUR/lectureBD_afficherNaissance.php?pr=${encodeURIComponent(prfctr)}`)
        .then(response => response.text())
        .then(data => {
            cible.innerHTML = data;
        })
        .catch(error => {
            console.error("Erreur AJAX :", error);
            cible.innerHTML = "Une erreur est survenue.";
        });
}
